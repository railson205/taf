CREATE TABLE usuarios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(50),
    data_nascimento DATE,
    sexo VARCHAR(10),
    matricula VARCHAR(50)
);

CREATE TABLE faixas_etarias (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome_grupo VARCHAR(15),
    idade_inicial INT,
    idade_final INT
);

CREATE TABLE exercicios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome_exercicio VARCHAR(40),
    modo_contagem VARCHAR(10)
);

CREATE TABLE notas (
    id INT PRIMARY KEY AUTO_INCREMENT,
    faixa_id INT,
    sexo VARCHAR(10),
    valor_nota FLOAT,
    exercicio_id INT,
    indice INT,
    FOREIGN KEY (faixa_id) REFERENCES faixas_etarias(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (exercicio_id) REFERENCES exercicios(id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE resultados (
    id INT PRIMARY KEY AUTO_INCREMENT,
    usuario_id INT,
    exercicio_id INT,
    nota_id INT,
    avaliador_id INT,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (exercicio_id) REFERENCES exercicios(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (nota_id) REFERENCES notas(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (avaliador_id) REFERENCES seguranca(id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

create TABLE seguranca(id int PRIMARY KEY AUTO_INCREMENT,
email varchar(50),
senha varchar(100),
nivel varchar(25),
usuario_id int,
FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE recuperacao_senhas (
  id INT UNSIGNED NOT NULL AUTO_INCREMENT,
  email VARCHAR(255) NOT NULL,
  token_hash CHAR(64) NOT NULL,     -- SHA-256 hex (opção segura)
  expires_at DATETIME NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  used TINYINT(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  UNIQUE KEY uk_email_token (email, token_hash(20)), -- evita duplicados curtos
  INDEX idx_token_hash (token_hash),
  INDEX idx_email (email),
  INDEX idx_expires_at (expires_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
