CREATE TABLE usuarios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(50),
    data_nascimento DATE,
    sexo VARCHAR(10),
    matricula VARCHAR(50)
);

CREATE TABLE faixas_etarias (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome_grupo VARCHAR(15),
    idade_inicial INT,
    idade_final INT
);

CREATE TABLE exercicios (
    id INT PRIMARY KEY AUTO_INCREMENT,
    nome_exercicio VARCHAR(40),
    modo_contagem VARCHAR(10)
);

CREATE TABLE notas (
    id INT PRIMARY KEY AUTO_INCREMENT,
    faixa_id INT,
    sexo VARCHAR(10),
    valor_nota FLOAT,
    exercicio_id INT,
    indice INT,
    FOREIGN KEY (faixa_id) REFERENCES faixas_etarias(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (exercicio_id) REFERENCES exercicios(id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE resultados (
    id INT PRIMARY KEY AUTO_INCREMENT,
    usuario_id INT,
    exercicio_id INT,
    nota_id INT,
    avaliador_id INT,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (exercicio_id) REFERENCES exercicios(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (nota_id) REFERENCES notas(id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (avaliador_id) REFERENCES seguranca(id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

create TABLE seguranca(id int PRIMARY KEY AUTO_INCREMENT, email varchar(50), senha varchar(50), usuario_id int, FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE ON UPDATE CASCADE);
