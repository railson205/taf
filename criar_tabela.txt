CREATE TABLE usuarios(id INT PRIMARY KEY AUTO_INCREMENT,nome varchar(50), data_nascimento date, sexo varchar(10));
CREATE TABLE faixas_etarias(id INT PRIMARY KEY AUTO_INCREMENT, nome_grupo varchar(15),idade_inicial int, idade_final int);
CREATE TABLE registro_exercicios(id INT PRIMARY KEY AUTO_INCREMENT, nome_exercicio varchar(40), tipo_exercicio varchar(10));
CREATE TABLE exercicios_usuarios(id INT PRIMARY KEY AUTO_INCREMENT,usuario_id int,registro_exercicio_id int,contagem_exercicio int, FOREIGN KEY (usuario_id) REFERENCES usuarios(id),FOREIGN KEY (registro_exercicio_id) REFERENCES registro_exercicios(id));
CREATE TABLE notas_exercicios(id INT PRIMARY KEY AUTO_INCREMENT, faixa_id INT, sexo VARCHAR(10),valor_nota float, registro_exercicio_id int, meta_exercicio int, FOREIGN KEY (faixa_id) REFERENCES faixas_etarias(id), FOREIGN KEY (registro_exercicio_id) REFERENCES registro_exercicios(id));
CREATE TABLE resultados_usuarios(id INT PRIMARY KEY AUTO_INCREMENT, usuario_id int, faixa_id int, registro_exercicio_id int,nota_ex_id int,exercicio_usuario_id int,FOREIGN KEY (exercicio_usuario_id) REFERENCES exercicios_usuarios(id), FOREIGN KEY (usuario_id) REFERENCES usuarios(id),FOREIGN KEY (faixa_id) REFERENCES faixas_etarias(id),FOREIGN KEY (registro_exercicio_id) REFERENCES registro_exercicios(id),FOREIGN KEY (nota_ex_id) REFERENCES notas_exercicios(id));
